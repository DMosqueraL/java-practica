// Debe: contar cuántos abonos pertenecen a la cuenta dada.
// Si la cuenta es nula, lanza IllegalArgumentException.
public static int contarPorCuenta(List<Abono> abonos, String cuenta) {
    if (cuenta == null) {
        throw new IllegalArgumentException("La cuenta no puede ser nula.");
    }
    int cantidad = 0;
    for (Abono abono : abonos) {
        if (abono.cuenta() == cuenta) {
            cantidad++;
        }
    }
    return cantidad;
}
