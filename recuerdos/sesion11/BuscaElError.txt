// Devuelve la fecha más reciente de una lista que no está vacía
static LocalDate masReciente(List<LocalDate> fechas) {
    LocalDate reciente = LocalDate.of(2026, 1, 1);
    for (LocalDate fecha : fechas) {
        if (fecha.isAfter(reciente)) {
            reciente = fecha;
        }
    }
    return reciente;
}
