public class CierreTema {

    static int dividir(int a, int b) {
        if (b == 0) {
            throw new IllegalArgumentException("b no puede ser 0");
        }
        return a / b;
    }

    static int repartir(int total, int n) {
        if (n <= 0) {
            throw new IllegalArgumentException("n invalido: " + n);
        }
        return total / n;
    }

    static int cuota(int total, int cuotas) {
        return repartir(total, cuotas);
    }

    static int largo(String texto) {
        return texto.length();
    }

    public static void main(String[] args) {
        System.out.println(dividir(10, 2));                       // [1]

        try {
            System.out.println("A");
            System.out.println(dividir(10, 0));
            System.out.println("B");                              // [2]
        } catch (IllegalArgumentException e) {
            System.out.println("C " + e.getMessage());            // [3]
        }
        System.out.println("D");                                  // [4]

        try {
            System.out.println(cuota(100, -1));
        } catch (IllegalArgumentException e) {
            System.out.println("E " + e.getMessage());            // [5]
        }

        try {
            System.out.println("F");
        } catch (IllegalArgumentException e) {
            System.out.println("G");                              // [6]
        }
        System.out.println("H");

        try {
            System.out.println(largo(null));                      // [7]
        } catch (IllegalArgumentException e) {
            System.out.println("I");
        }
        System.out.println("J");                                  // [8]
    }
}
// --- Predicciones
[1] 5 ---> Realiza la división entera normal, el dividendo 2 es diferente de 0
[2] A ---> Es la primera línea del try, aún no se ha lanzado la excepción por división entera por cero.
           No se imprime B, pq está dentro del try y es la siguiente línea donde se lanza la excepción, se corta ahí (se salta el resto de las líneas de try)
[3] C b no puede ser 0 --> Muestra el texto que imprime el bloque catch y el mensaje la excepción IllegalArgumentException que coincide con el catch
[4] D ---> Se imprime normal ya que está por fuera del bloque try/catch y el programa sigue después del bloque
[5] E n invalido: -1 ---> Se lanza la excepción IllegalArgumentException que coincide con el catch
[6] F ---> Se imprime la cadena que está dentro del println. No hay ningún llamado a algún método
[7] NullPointerException ---> La JVM imprime el stack trace (tipo, mensaje y la cadena de llamadas (at ...)) dado que nadie atrapa la excepción por null y main termina
[8] No se imprime pq main terminó en el bloque anterior
