// Debe: aplicar un descuento de 0 a 100 por ciento (ambos incluidos) y devolver el precio final.
// Si el porcentaje está fuera de ese rango, lanza IllegalArgumentException.
public static int precioConDescuento(int precio, int porcentaje) {
    if (porcentaje < 0 || porcentaje >= 100) {
        throw new IllegalArgumentException("El porcentaje debe estar entre 0 y 100 y llegó " + porcentaje);
    }
    return precio - precio * porcentaje / 100;
}
